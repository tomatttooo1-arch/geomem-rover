cmake_minimum_required(VERSION 3.10)
project(planner_model LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(planner_model SHARED model.cpp model_v2.cpp model_v3.cpp vendor/navfn.cpp)
target_include_directories(planner_model PRIVATE vendor)
target_compile_options(planner_model PRIVATE -Wall -Wextra -Wno-unused-parameter -ffp-contract=off)
add_executable(planner_model_tests tests.cpp)
target_link_libraries(planner_model_tests planner_model)
enable_testing()
add_test(NAME planner_model_controls COMMAND planner_model_tests)

add_executable(planner_v2_tests tests_v2.cpp)
target_link_libraries(planner_v2_tests planner_model)
add_test(NAME planner_v2_controls COMMAND planner_v2_tests)

add_executable(planner_v3_tests tests_v3.cpp)
target_link_libraries(planner_v3_tests planner_model)
add_test(NAME planner_v3_controls COMMAND planner_v3_tests)

add_library(planner_realpose_navfn SHARED realpose_navfn.cpp)
target_include_directories(planner_realpose_navfn PRIVATE vendor)
target_link_libraries(planner_realpose_navfn planner_model)
target_compile_options(planner_realpose_navfn PRIVATE -Wall -Wextra -ffp-contract=off)

# Host-only verification bridge. The deliverable header has no third-party deps.
add_library(planner_hwref SHARED hwref/bridge.cpp)
target_compile_options(planner_hwref PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
add_executable(planner_hwref_tests hwref/tests.cpp)
target_compile_options(planner_hwref_tests PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
add_test(NAME planner_hwref_controls COMMAND planner_hwref_tests)
add_executable(planner_hwref_replay hwref/replay.cpp)
target_compile_options(planner_hwref_replay PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
